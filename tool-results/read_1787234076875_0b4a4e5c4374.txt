     6→     6→function shuffle<T>(arr: T[]): T[] {
     7→     7→  const a = [...arr];
     8→     8→  for (let i = a.length - 1; i > 0; i--) {
     9→     9→    const j = Math.floor(Math.random() * (i + 1));
    10→    10→    [a[i], a[j]] = [a[j], a[i]];
    11→    11→  }
    12→    12→  return a;
    13→    13→}

    15→    15→function nextPow2(n: number): number {
    16→    16→  let p = 1;
    17→    17→  while (p < n) p *= 2;
    18→    18→  return p;
    19→    19→}
